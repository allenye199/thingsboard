要在ThingsBoard的Asset Table Widget中实现自动创建基于当前资产的子资产和设备，你需要创建一个自定义操作按钮，并编写相应的脚本。以下是一个详细的实施方案：

```javascript
/*
 * Asset Table Widget的自定义动作脚本
 * 功能：为当前资产创建子资产和关联设备
 */

// 获取必要的服务
var $injector = widgetContext.$scope.$injector;
var assetService = $injector.get(widgetContext.servicesMap.get('assetService'));
var deviceService = $injector.get(widgetContext.servicesMap.get('deviceService'));
var entityRelationService = $injector.get(widgetContext.servicesMap.get('entityRelationService'));
var dialogs = $injector.get(widgetContext.servicesMap.get('dialogs'));
var attributeService = $injector.get(widgetContext.servicesMap.get('attributeService'));

// 从页面上下文获取当前租户ID
var currentTenantId = widgetContext.currentUser.tenantId;

// 创建子资产和设备的主函数
function createChildEntities() {
  // 显示确认对话框
  var title = '创建确认';
  var content = '是否要为 "' + entityName + '" 创建子资产和设备？';
  
  dialogs.confirm(title, content, '取消', '确定').subscribe(
    function(result) {
      if (result) {
        // 用户确认后，开始创建
        performCreation();
      }
    }
  );
}

// 执行实际的创建操作
function performCreation() {
  // 1. 首先创建子资产 - 根据当前资产类型决定子资产类型
  var parentType = entityId.entityType;
  var parentId = entityId.id;
  var childAssetType = determineChildAssetType(parentType);
  
  if (childAssetType) {
    createChildAsset(parentId, childAssetType).then(
      function(childAsset) {
        // 2. 子资产创建成功后，创建对应的设备
        return createAssociatedDevice(childAsset);
      }
    ).then(
      function(result) {
        // 3. 创建实体间的关系
        return createRelationships(result.asset, result.device);
      }
    ).then(
      function() {
        // 4. 全部操作成功，刷新UI
        widgetContext.updateAliases();
        widgetContext.showSuccessToast('子资产和设备已成功创建!');
      }
    ).catch(
      function(error) {
        console.error('创建过程失败:', error);
        widgetContext.showErrorToast('创建失败: ' + (error.data ? error.data.message : '未知错误'));
      }
    );
  } else {
    widgetContext.showErrorToast('无法确定子资产类型!');
  }
}

// 根据父资产类型确定子资产类型
function determineChildAssetType(parentType) {
  // 根据你的业务逻辑实现资产类型映射
  var typeMapping = {
    'TENANT': 'CUSTOMER',
    'CUSTOMER': 'FARM',
    'FARM': 'UNIT',
    'UNIT': 'PEN',
    'ASSET': 'ASSET'  // 默认情况
  };
  
  return typeMapping[parentType] || 'ASSET';
}

// 创建子资产
function createChildAsset(parentId, assetType) {
  var assetName = entityName + "_子资产_" + new Date().getTime();
  var assetLabel = entityName + " 的子资产";
  
  // 准备资产数据
  var assetData = {
    name: assetName,
    type: assetType,
    label: assetLabel
  };
  
  // 创建资产
  return assetService.saveAsset(assetData).toPromise();
}

// 创建关联设备
function createAssociatedDevice(asset) {
  var deviceName = asset.name + "_设备";
  var deviceType = "默认设备";
  var deviceLabel = asset.label + " 关联设备";
  
  // 准备设备数据
  var deviceData = {
    name: deviceName,
    type: deviceType,
    label: deviceLabel
  };
  
  // 创建设备并返回包含资产和设备的结果
  return deviceService.saveDevice(deviceData).toPromise()
    .then(function(device) {
      return {
        asset: asset,
        device: device
      };
    });
}

// 创建实体间的关系
function createRelationships(asset, device) {
  // 创建父资产与子资产的关系
  var parentToChildRelation = {
    from: entityId,
    to: {
      entityType: 'ASSET',
      id: asset.id.id
    },
    type: 'Contains'
  };
  
  // 创建子资产与设备的关系
  var assetToDeviceRelation = {
    from: {
      entityType: 'ASSET',
      id: asset.id.id
    },
    to: {
      entityType: 'DEVICE',
      id: device.id.id
    },
    type: 'Contains'
  };
  
  // 创建所有关系
  var promises = [
    entityRelationService.saveRelation(parentToChildRelation).toPromise(),
    entityRelationService.saveRelation(assetToDeviceRelation).toPromise()
  ];
  
  return Promise.all(promises);
}

// 为子资产和设备设置初始属性
function setInitialAttributes(asset, device) {
  // 设置子资产的属性
  var assetAttributes = {
    'createdTime': new Date().getTime(),
    'parentId': entityId.id
  };
  
  // 设置设备的属性
  var deviceAttributes = {
    'createdTime': new Date().getTime(),
    'assetId': asset.id.id
  };
  
  var promises = [
    attributeService.saveEntityAttributes(
      { entityType: 'ASSET', id: asset.id.id },
      'SERVER_SCOPE',
      assetAttributes
    ).toPromise(),
    attributeService.saveEntityAttributes(
      { entityType: 'DEVICE', id: device.id.id },
      'SERVER_SCOPE',
      deviceAttributes
    ).toPromise()
  ];
  
  return Promise.all(promises);
}

// 执行主函数
createChildEntities();
```

## 使用指南

1. **创建自定义操作按钮**:
   
   打开资产表格小部件配置，在"操作"标签页中添加一个自定义操作按钮：
   
   - 名称: "创建子资产和设备"
   - 图标: "add_circle_outline"
   - 动作类型: "自定义"
   - 将上述JavaScript代码复制到自定义动作代码区域

2. **业务逻辑定制**:
   
   根据你的具体业务需求修改以下函数:
   
   - `determineChildAssetType()`: 定义父资产类型与子资产类型的映射关系
   - `createChildAsset()`: 自定义子资产的命名和属性
   - `createAssociatedDevice()`: 自定义设备的命名和属性
   - `setInitialAttributes()`: 设置初始属性值

3. **权限配置**:
   
   确保用户有权限创建资产和设备。登录用户需要具有以下权限:
   
   - 资产: 创建、读取
   - 设备: 创建、读取
   - 关系: 创建

该脚本会在用户点击按钮并确认后:

1. 为当前资产创建一个子资产（类型根据父资产类型确定）
2. 为新创建的子资产创建一个关联设备
3. 建立父资产与子资产、子资产与设备之间的关系
4. 设置必要的初始属性

这样，你就可以在资产表格中一键创建完整的资产-设备层级结构，极大提高配置效率。
